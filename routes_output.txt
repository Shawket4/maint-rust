--- src/handlers/cache_invoices.rs ---
pub fn routes() -> Scope {
    // Order matters: more-specific paths must register before less-specific ones,
    // otherwise actix's path matcher will route /sync to the parameterized handler.
    web::scope("")
        .service(sync_invoices)
        .service(search_invoices_handler)
        .service(get_invoices)
}

--- src/handlers/tires.rs ---
pub fn routes() -> Scope {
    // Bulk path must register before the parameterized /tires/{id} so it isn't shadowed.
    web::scope("")
        .service(tires_bulk)
        .service(mount_handler)
        .service(dismount_handler)
        .service(create_tire)
        .service(list_tires)
        .service(tire_history)
        .service(get_tire)
        .service(update_tire)
        .service(delete_tire)
}
